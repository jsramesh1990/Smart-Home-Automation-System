#!/bin/bash

# Build Desktop Controller
echo "Building SmartLock Pro Desktop Controller..."
mkdir -p build
cd build
cmake ..
make

# Build ESP32 Firmware
echo "Building ESP32 Firmware..."
cd ../firmware
platformio run

# Create installation directory
echo "Creating installation..."
mkdir -p ../install/bin
cp ../build/smartlock_controller ../install/bin/
cp -r ../firmware/.pio/build/esp32dev/firmware.bin ../install/

echo "Build complete!"
echo "Desktop controller: ./install/bin/smartlock_controller"
echo "ESP32 firmware: ./install/firmware.bin"


Systemd Service File
/etc/systemd/system/smartlock.service

[Unit]
Description=SmartLock Pro Controller
After=network.target sqlite.service

[Service]
Type=simple
User=smartlock
WorkingDirectory=/opt/smartlock
ExecStart=/opt/smartlock/smartlock_controller
Restart=on-failure
RestartSec=10
StandardOutput=journal
StandardError=journal

[Install]
WantedBy=multi-user.target
